function confetti(count) {
  const colors = ['#ff8fab','#ffd166','#8ce8c4','#b8a7f9','#ffb3c6'];
  for (let i = 0; i < (count || 60); i++) {
    const c = document.createElement('div');
    c.className = 'confetti';
    c.style.left = Math.random() * 100 + 'vw';
    c.style.background = colors[i % colors.length];
    c.style.width = (8 + Math.random() * 8) + 'px';
    c.style.height = (10 + Math.random() * 8) + 'px';
    c.style.animationDelay = (Math.random() * 1.5) + 's';
    c.style.animationDuration = (2.5 + Math.random() * 2) + 's';
    document.body.appendChild(c);
    setTimeout(function(){ c.remove(); }, 6000);
  }
}

function renderVocab(key) {
  const grid = document.getElementById('vocab-' + key);
  if (!grid || grid.dataset.done) return;
  grid.dataset.done = '1';
  VOCAB[key].forEach(function(v) {
    const el = document.createElement('div');
    el.className = 'flip';
    el.innerHTML =
      '<div class="face front"><div class="emoji">' + v.e + '</div><div class="hanzi">' + v.h + '</div></div>' +
      '<div class="face back"><div class="pinyin">' + v.p + '</div><div class="hanzi small">' + v.h + '</div><div class="ru">' + v.r + '</div></div>';
    el.addEventListener('click', function(){ el.classList.toggle('flipped'); });
    grid.appendChild(el);
  });
}

/* ================= QUIZ ================= */
function renderQuiz(key) {
  const box = document.getElementById('quiz-' + key);
  const qs = QUIZZES[key];
  let i = 0;
  function showQ() {
    if (i >= qs.length) {
      box.innerHTML = '<div class="quiz-done">Тема пройдена! 🎉<br>Ты молодец! 🐼</div>';
      confetti(30);
      return;
    }
    const q = qs[i];
    box.innerHTML =
      '<div class="q-num">Вопрос ' + (i + 1) + ' из ' + qs.length + '</div>' +
      '<div class="q-text">' + q.q + '</div>' +
      '<div class="q-opts"></div>' +
      '<div class="q-feedback"></div>';
    const opts = box.querySelector('.q-opts');
    const fb = box.querySelector('.q-feedback');
    let answered = false;
    q.opts.forEach(function(opt, idx) {
      const b = document.createElement('button');
      b.className = 'opt';
      b.textContent = opt;
      b.addEventListener('click', function() {
        if (answered) return;
        if (idx === q.a) {
          answered = true;
          b.classList.add('correct');
          Array.prototype.forEach.call(opts.children, function(o){ o.disabled = true; });
          addStar(1);
          fb.innerHTML = 'Правильно! 🎉';
          const nb = document.createElement('button');
          nb.className = 'btn next-q';
          nb.textContent = (i === qs.length - 1) ? 'Завершить ⭐' : 'Дальше ▶';
          nb.addEventListener('click', function(){ i++; showQ(); });
          fb.appendChild(nb);
        } else {
          b.classList.add('wrong');
          b.disabled = true;
          fb.innerHTML = 'Попробуй ещё раз! 💪';
        }
      });
      opts.appendChild(b);
    });
  }
  showQ();
}

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    const t = arr[i]; arr[i] = arr[j]; arr[j] = t;
  }
  return arr;
}

function renderMatch() {
  const grid = document.getElementById('matchGrid');
  const cards = [];
  PAIRS.forEach(function(pr, idx) {
    cards.push({ pair: idx, text: pr[0], type: 'emoji' });
    cards.push({ pair: idx, text: pr[1], type: 'hanzi' });
  });
  shuffle(cards);
  let open = [], lock = false, matched = 0;
  grid.innerHTML = '';
  cards.forEach(function(c) {
    const d = document.createElement('div');
    d.className = 'mcard ' + c.type;
    d.dataset.pair = c.pair;
    d.innerHTML = '<span class="m-inner">' + c.text + '</span>';
    d.addEventListener('click', function() {
      if (lock || d.classList.contains('open') || d.classList.contains('matched')) return;
      d.classList.add('open');
      open.push(d);
      if (open.length === 2) {
        lock = true;
        const a = open[0], b = open[1];
        if (a.dataset.pair === b.dataset.pair) {
          setTimeout(function() {
            a.classList.remove('open'); b.classList.remove('open');
            a.classList.add('matched'); b.classList.add('matched');
            matched++; open = []; lock = false;
            if (matched === PAIRS.length) {
              addStar(2);
              document.getElementById('matchWin').style.display = 'block';
              confetti(80);
            }
          }, 400);
        } else {
          setTimeout(function() {
            a.classList.remove('open'); b.classList.remove('open');
            open = []; lock = false;
          }, 850);
        }
      }
    });
    grid.appendChild(d);
  });
}

/* ================= NAVIGATION ================= */
const dotsEl = document.getElementById('dots');
slides.forEach(function(_, i) {
  const d = document.createElement('span');
  d.className = 'dot' + (i === 0 ? ' on' : '');
  d.addEventListener('click', function(){ go(i); });
  dotsEl.appendChild(d);
});

function updateFinal() {
  document.getElementById('finalStars').textContent = stars;
  let msg;
  if (stars >= 15) msg = 'Супер! Ты настоящий мастер китайского! 🏆';
  else if (stars >= 10) msg = 'Отличная работа! Продолжай в том же духе! 🌟';
  else msg = 'Хорошо! Повтори урок ещё раз — будет ещё лучше! 💪';
  document.getElementById('finalMsg').textContent = msg;
}

function activate(slide) {
  if (slide.dataset.vocab) renderVocab(slide.dataset.vocab);
  if (slide.dataset.quiz && !slide.dataset.done) { slide.dataset.done = '1'; renderQuiz(slide.dataset.quiz); }
  if (slide.dataset.match && !slide.dataset.done) { slide.dataset.done = '1'; renderMatch(); }
  if (slide.dataset.final) updateFinal();
  document.getElementById('progressFill').style.width = (cur / (slides.length - 1) * 100) + '%';
}

function go(i) {
  if (i < 0 || i >= slides.length) return;
  slides[cur].classList.remove('active');
  cur = i;
  slides[cur].classList.add('active');
  Array.prototype.forEach.call(dotsEl.children, function(d, j) {
    d.classList.toggle('on', j === cur);
  });
  activate(slides[cur]);
  window.scrollTo(0, 0);
}

document.getElementById('prevBtn').addEventListener('click', function(){ go(cur - 1); });
document.getElementById('nextBtn').addEventListener('click', function(){ go(cur + 1); });
document.getElementById('startBtn').addEventListener('click', function(){ go(1); });

document.addEventListener('keydown', function(e) {
  if (e.key === 'ArrowRight') go(cur + 1);
  if (e.key === 'ArrowLeft') go(cur - 1);
});
